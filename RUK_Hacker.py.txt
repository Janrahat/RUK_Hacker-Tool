# ==================== URL INFO ====================
def url_info():
    print(f"\n{C}[{G}+{C}] Target URL Information{RESET}")
    url = input(f"{Y}[?]{W} Enter Target URL: ").strip()
    
    if not url.startswith(('http://', 'https://')):
        url = 'https://' + url
    
    print(f"\n{C}[{G}*]{C} Gathering information...{RESET}\n")
    time.sleep(1)
    
    try:
        r = requests.get(url, timeout=10)
        print(f"{G}[+] Status Code     : {W}{r.status_code}")
        print(f"{G}[+] Server          : {W}{r.headers.get('Server', 'N/A')}")
        print(f"{G}[+] Content-Type    : {W}{r.headers.get('Content-Type', 'N/A')}")
        print(f"{G}[+] Content-Length  : {W}{r.headers.get('Content-Length', 'N/A')}")
        print(f"{G}[+] Date            : {W}{r.headers.get('Date', 'N/A')}")
        
        # IP & Domain Info
        domain = url.split('//')[1].split('/')[0]
        ip = socket.gethostbyname(domain)
        print(f"{G}[+] Domain          : {W}{domain}")
        print(f"{G}[+] IP Address      : {W}{ip}")
        
    except Exception as e:
        print(f"{R}[-] Error: {str(e)}{RESET}")
    
    input(f"\n{C}[{Y}Press Enter to continue{RESET}]")
